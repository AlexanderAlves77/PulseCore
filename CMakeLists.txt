cmake_minimum_required(VERSION 3.16)

project(PulseCore VERSION 0.1 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

find_package(Qt6 REQUIRED COMPONENTS Quick Quick3D Test)

qt_standard_project_setup(REQUIRES 6.10)

include(CTest)

# Content of downloaded design from Figma to Qt or similar
if (EXISTS "${CMAKE_CURRENT_LIST_DIR}/importedcontent/CMakeLists.txt")
    add_subdirectory(${CMAKE_CURRENT_LIST_DIR}/importedcontent)
endif()

qt_add_executable(appPulseCore
    main.cpp

    Core/Enums/SimulationState.h
    Core/Enums/SimulationScenario.h

    Core/Models/VitalSigns.h

    Core/Simulation/SimulationConfig.h
    Core/Simulation/SimulationScenarioFactory.h

    Core/Simulation/SimulationController.h
    Core/Simulation/SimulationController.cpp

    Core/Simulation/VitalSignsGenerator.h
    Core/Simulation/VitalSignsGenerator.cpp

    Core/Simulation/SimulationClock.h
    Core/Simulation/SimulationClock.cpp

    Core/Simulation/EcgWaveformGenerator.h
    Core/Simulation/EcgWaveformGenerator.cpp
)

target_include_directories(appPulseCore
    PRIVATE
        ${CMAKE_CURRENT_SOURCE_DIR}
)

# QML components are exposed at the PulseCore module root.
set_source_files_properties(
    Heart3D/Heart3D.qml
    PROPERTIES QT_RESOURCE_ALIAS Heart3D.qml
)

set_source_files_properties(
    Heart3D/Imported/Heart.qml
    PROPERTIES QT_RESOURCE_ALIAS Heart.qml
)

# Preserve the relative resource paths generated by Balsam.
set_source_files_properties(
    Heart3D/Imported/meshes/mesh_mesh.mesh
    PROPERTIES QT_RESOURCE_ALIAS meshes/mesh_mesh.mesh
)

set_source_files_properties(
    Heart3D/Imported/maps/textureData.jpg
    PROPERTIES QT_RESOURCE_ALIAS maps/textureData.jpg
)

set_source_files_properties(
    Heart3D/Imported/maps/textureData7.jpg
    PROPERTIES QT_RESOURCE_ALIAS maps/textureData7.jpg
)

qt_add_qml_module(appPulseCore
    URI PulseCore

        Components/Monitoring/CompactVitalCard.qml
        Components/Monitoring/RightMonitoringPanel.qml
        Components/Monitoring/LeftMonitoringPanel.qml

# Qt for iOS sets MACOSX_BUNDLE_GUI_IDENTIFIER automatically since Qt 6.1.
# If you are developing for iOS or macOS you should consider setting an
# explicit, fixed bundle identifier manually though.
set_target_properties(appPulseCore PROPERTIES
#    MACOSX_BUNDLE_GUI_IDENTIFIER com.example.appPulseCore
    MACOSX_BUNDLE_BUNDLE_VERSION ${PROJECT_VERSION}
    MACOSX_BUNDLE_SHORT_VERSION_STRING ${PROJECT_VERSION_MAJOR}.${PROJECT_VERSION_MINOR}
    MACOSX_BUNDLE TRUE
    WIN32_EXECUTABLE TRUE
)

target_link_libraries(appPulseCore
    PRIVATE
        Qt6::Quick
        Qt6::Quick3D
)

install(TARGETS appPulseCore
    BUNDLE DESTINATION .
    LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
    RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR}
)
